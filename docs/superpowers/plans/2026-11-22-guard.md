# Guard (PreToolUse interception) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `memai hook guard` — a PreToolUse hook that denies whole-file reads above a measured threshold and returns the index's signature page with concrete next moves, overridable in one step, on Claude Code and ZCode.

**Architecture:** Pure decision logic in a new `src/guard.mjs` (config, glob matching, payload extraction, cost rule, override state). `mcp/hook.mjs` gains `hookGuardPayload()` which resolves the project and store and delegates; `bin/memai.mjs` gains a `hook guard` stdin verb that never fails the session. Adapters ship the verified config shapes. `bench/guard-sim.mjs` validates the 8 KB default on the existing task corpora.

**Tech Stack:** plain ESM `.mjs`, Node ≥ 22.13, `node:sqlite` via the existing store, no dependencies. Tests: `node --test "test/*.test.mjs"`.

**Spec:** [`docs/superpowers/specs/2026-11-22-guard-design.md`](../specs/2026-11-22-guard-design.md).

## Global Constraints

- **Exit 0 always** from every hook path; a guard bug must never break a session.
- **stdout is empty or exactly one JSON object**; diagnostics on stderr only when `MEMAI_DEBUG=1`.
- **No new dependencies**, no schema changes, no new MCP ops.
- **Read-only on the allow path**: no state-file writes unless denying.
- **Decision order is fixed** (tests depend on it): enabled → tool name → path → inside root → whole-file → allowlist → exists → minBytes → indexed → fresh → page → cost rule → override.
- Version bump to **0.5.0** in exactly three places: `package.json`, `src/mem-cli.mjs` `VERSION`, `mcp/paths.mjs` `VERSION`.
- Token estimator: `estTokens` from `src/tokens.mjs`; file tokens in the rule use `Math.ceil(bytes / 3.966)` per the spec.

```js
// Guard: PreToolUse interception of whole-file reads.
//
// The decision logic lives here so the hook (mcp/hook.mjs), the CLI status
// command, and the bench all share one implementation. Design:
// docs/superpowers/specs/2026-11-22-guard-design.md
//
// Rules: deny only when a whole read costs at least twice the paged
// alternative; never throw into the caller (a hook must not break a session).
import fs from 'node:fs'
import path from 'node:path'

export const GUARD_DEFAULTS = Object.freeze({
  enabled: true,
  minBytes: 8192,
  allow: [],
  toolNames: ['Read', 'read_file'],
})
export const SIG_BUDGET = 400        // token budget of the signature excerpt
export const FAULT_EST_TOKENS = 400  // one typical fault slice
export const STATE_TTL_MS = 3600_000 // override lifetime

/** Effective config: guard.json, then env overrides. Never throws. */
export function loadGuardConfig({ root, env = process.env } = {}) {
  const cfg = { enabled: GUARD_DEFAULTS.enabled, minBytes: GUARD_DEFAULTS.minBytes, allow: [], toolNames: [...GUARD_DEFAULTS.toolNames], source: 'defaults' }
  if (root) {
    try {
      const raw = JSON.parse(fs.readFileSync(path.join(root, '.memai', 'guard.json'), 'utf8'))
      cfg.source = 'guard.json'
      if (typeof raw?.enabled === 'boolean') cfg.enabled = raw.enabled
      if (Number.isFinite(raw?.minBytes) && raw.minBytes > 0) cfg.minBytes = Math.trunc(raw.minBytes)
      if (Array.isArray(raw?.allow)) cfg.allow = raw.allow.filter((g) => typeof g === 'string')
      if (Array.isArray(raw?.toolNames) && raw.toolNames.length) cfg.toolNames = raw.toolNames.filter((t) => typeof t === 'string')
    } catch { /* missing or invalid guard.json -> defaults */ }
  }
  if (String(env.MEMAI_GUARD ?? '').toLowerCase() === 'off') { cfg.enabled = false; cfg.source = 'env' }
  const envMin = Number(env.MEMAI_GUARD_MIN_BYTES)
  if (Number.isFinite(envMin) && envMin > 0) { cfg.minBytes = Math.trunc(envMin); cfg.source = 'env' }
  return cfg
}

/** Glob with `**` (any depth), `*` (one segment), `?` (one char). */
export function globMatch(pattern, rel) {
  const p = String(pattern).replace(/\\/g, '/')
  let re = '^'
  for (let i = 0; i < p.length; i++) {
    const c = p[i]
    if (c === '*') {
      if (p[i + 1] === '*') { re += '.*'; i++ } else re += '[^/]*'
    } else if (c === '?') re += '[^/]'
    else re += c.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
  }
  return new RegExp(re + '$').test(String(rel).replace(/\\/g, '/'))
}

export function guardToolName(payload) {
  const v = payload?.tool_name ?? payload?.toolName ?? payload?.name ?? ''
  return String(v ?? '')
}

/** File path from Claude Code (`tool_input.file_path`) or ZCode-ish shapes. */
export function guardFilePath(payload) {
  const ti = payload?.tool_input ?? payload?.toolInput ?? payload?.inputs ?? {}
  const v = ti?.file_path ?? ti?.filePath ?? ti?.path ?? ti?.file ?? payload?.file_path
  return typeof v === 'string' && v.trim() ? v.trim() : null
}

/** A read with any range parameter is partial — never guarded. */
export function isWholeFileRead(toolInput = {}) {
  const keys = ['offset', 'limit', 'start_line', 'end_line', 'startLine', 'endLine', 'start', 'end']
  for (const k of keys) {
    const v = toolInput?.[k]
    if (v !== undefined && v !== null && v !== '') return false
  }
  return true
}
```

```js
// ---------------------------------------------------------------- state ---

export function statePath(root) { return path.join(root, '.memai', 'guard-state.json') }

/** Deny/override state. Corrupt or missing file reads as empty. */
export function readState(root) {
  try {
    const raw = JSON.parse(fs.readFileSync(statePath(root), 'utf8'))
    const overrides = raw?.overrides && typeof raw.overrides === 'object' ? raw.overrides : {}
    return { version: 1, overrides, denyCount: Number(raw?.denyCount) || 0, lastDeny: raw?.lastDeny ?? null }
  } catch {
    return { version: 1, overrides: {}, denyCount: 0, lastDeny: null }
  }
}

/** Atomic write (tmp + rename); callers already swallow failures. */
export function writeState(root, state) {
  const p = statePath(root)
  fs.mkdirSync(path.dirname(p), { recursive: true })
  const tmp = `${p}.${process.pid}.tmp`
  fs.writeFileSync(tmp, JSON.stringify(state))
  fs.renameSync(tmp, p)
}

export function hasOverride(root, { sessionId, rel, now = Date.now(), ttlMs = STATE_TTL_MS }) {
  const ts = readState(root).overrides[`${sessionId}|${rel}`]
  return typeof ts === 'number' && now - ts < ttlMs
}

/** Record a deny and the one-shot override it grants for this session. */
export function recordDeny(root, { sessionId, rel, now = Date.now(), bytes = 0, fileTokens = 0, sigTokens = 0, ttlMs = STATE_TTL_MS }) {
  const st = readState(root)
  st.overrides[`${sessionId}|${rel}`] = now
  for (const [k, ts] of Object.entries(st.overrides)) {
    if (!(typeof ts === 'number' && now - ts < ttlMs)) delete st.overrides[k]
  }
  st.denyCount += 1
  st.lastDeny = { rel, at: new Date(now).toISOString(), bytes, fileTokens, sigTokens }
  try { writeState(root, st) } catch { /* a guard must never fail the session */ }
}

// ------------------------------------------------------------- decision ---

const kb = (n) => (n / 1024).toFixed(1)

/** The deny reason: size + cost, the signature page, and exactly what to do next. */
export function buildDenyReason({ rel, bytes, fileTokens, sigText, sigTokens, minBytes, store }) {
  const syms = (store?.symbolsForPage?.(`file:${rel}`) ?? [])
    .filter((s) => !['heading', 'test', 'import', 'key'].includes(s.kind))
    .slice(0, 3)
  const lines = [
    `# MemAi guard — ${rel} withheld (${kb(bytes)} KB ≈ ${fileTokens} tok)`,
    `Whole-file reads above ${kb(minBytes)} KB return signatures first; Read this path again to override for this session.`,
    `Signatures (index page, ${sigTokens} tok):`,
    String(sigText ?? '').trim(),
    'Next moves:',
    ...syms.map((s, i) => `  fault ${s.name}${i === 0 ? '' : ' --deep'}   # ~300–600 tok`),
    `  Read ${rel} again   # override`,
  ]
  return lines.join('\n')
}

/**
 * The whole decision. Order is contractual (tests assert reasons):
 * enabled -> tool -> path -> root -> whole-file -> allowlist -> exists ->
 * minBytes -> indexed -> fresh -> page -> cost rule -> override.
 */
export function decideGuard({ payload = null, root, env = process.env, cwd = process.cwd(), now = Date.now(), store, pages } = {}) {
  const cfg = loadGuardConfig({ root, env })
  if (!cfg.enabled) return { decision: 'allow', reason: 'guard disabled' }
  if (!cfg.toolNames.includes(guardToolName(payload))) return { decision: 'allow', reason: 'not a guarded read tool' }
  const raw = guardFilePath(payload)
  if (!raw) return { decision: 'allow', reason: 'no file path in payload' }
  const abs = path.resolve(cwd, raw)
  const rel = path.relative(root, abs).split(path.sep).join('/')
  if (!rel || rel.startsWith('..') || path.isAbsolute(rel)) return { decision: 'allow', reason: 'outside project root' }
  const ti = payload?.tool_input ?? payload?.toolInput ?? payload?.inputs ?? {}
  if (!isWholeFileRead(ti)) return { decision: 'allow', reason: 'partial read' }
  if (cfg.allow.some((g) => globMatch(g, rel))) return { decision: 'allow', reason: 'allowlisted' }

  let st
  try { st = fs.statSync(abs) } catch { return { decision: 'allow', reason: 'file missing' } }
  if (st.size < cfg.minBytes) return { decision: 'allow', reason: 'below minBytes' }

  const row = store.getFile(rel)
  if (!row) return { decision: 'allow', reason: 'not indexed' }
  try {
    if (sha1(fs.readFileSync(abs)) !== row.content_hash) return { decision: 'allow', reason: 'stale index' }
  } catch { return { decision: 'allow', reason: 'file unreadable' } }

  const sig = pages.get(`file:${rel}`, { budget: SIG_BUDGET })
  if (!sig?.ok) return { decision: 'allow', reason: 'no page' }
  const fileTokens = Math.ceil(st.size / 3.966)
  if (fileTokens < 2 * (sig.tokens + FAULT_EST_TOKENS)) return { decision: 'allow', reason: 'paging would not halve the read' }

  const sessionId = String(payload?.session_id ?? payload?.sessionId ?? 'anon')
  if (hasOverride(root, { sessionId, rel, now })) return { decision: 'allow', reason: 'override' }

  recordDeny(root, { sessionId, rel, now, bytes: st.size, fileTokens, sigTokens: sig.tokens })
  const text = buildDenyReason({ rel, bytes: st.size, fileTokens, sigText: sig.text, sigTokens: sig.tokens, minBytes: cfg.minBytes, store })
  return { decision: 'deny', rel, bytes: st.size, fileTokens, sigTokens: sig.tokens, text }
}
```

```js
 * proceed — the caller must stay silent in that case. Never throws.
 */
export function hookGuardPayload({ payload = null, root = null, db = null, env = process.env, cwd = process.cwd(), now = Date.now() } = {}) {
  const filePath = guardFilePath(payload)
  if (!filePath) return { json: null, decision: 'allow', reason: 'no file path in payload' }
  const abs = path.resolve(cwd, filePath)
  const projectRoot = root ? path.resolve(root) : nearestProjectRoot(path.dirname(abs))
  const dbPath = resolveDb(projectRoot, { db, env })
  if (!fs.existsSync(dbPath)) return { json: null, decision: 'allow', reason: `no store at ${dbPath}` }
  const store = new MemoryStore({ dbPath, root: projectRoot })
  try {
    const pages = new MemoryPages({ store, root: projectRoot })
    const r = decideGuard({ payload, root: projectRoot, env, cwd, now, store, pages })
    if (r.decision !== 'deny') return { json: null, decision: 'allow', reason: r.reason }
    return {
      json: { hookSpecificOutput: { hookEventName: 'PreToolUse', permissionDecision: 'deny', permissionDecisionReason: r.text } },
      decision: 'deny', text: r.text, tokens: estTokens(r.text),
    }
  } finally {
    store.close()
  }
}
```

```js
// Guard threshold simulation: does the 8 KB default hold up on the benchmark
// corpora? For every file the 14 tasks touch, compare a whole-file read with
// the guard-forced alternative (signature page + one fault estimate).
//
// Run: node bench/guard-sim.mjs
import fs from 'node:fs'
import path from 'node:path'
import url from 'node:url'
import { corpusSpecs, tasksFor, checkCorpus } from './corpora.mjs'
import { MemoryStore } from '../src/store.mjs'
import { MemoryPages } from '../src/page.mjs'
import { estTokens } from '../src/tokens.mjs'
import { SIG_BUDGET, FAULT_EST_TOKENS } from '../src/guard.mjs'

const HERE = path.dirname(url.fileURLToPath(import.meta.url))
const TOOL_OVERHEAD = 45
export const THRESHOLDS = [4096, 8192, 16384]

function freshDb(id) {
  const dir = path.join(HERE, '.tmp')
  fs.mkdirSync(dir, { recursive: true })
  const p = path.join(dir, `guard-${id}.db`)
  for (const suf of ['', '-wal', '-shm']) fs.rmSync(p + suf, { force: true })
  return p
}

export async function guardSim({ specIds = null, thresholds = THRESHOLDS } = {}) {
  const rows = []
  for (const spec of corpusSpecs()) {
    if (specIds && !specIds.includes(spec.id)) continue
    if (!checkCorpus(spec).ok) continue
    const dbPath = freshDb(spec.id)
    const store = new MemoryStore({ dbPath, root: spec.root })
    const pages = new MemoryPages({ store, root: spec.root })
    try {
      await pages.index({ maxFiles: 5000, ignore: spec.ignore ?? [] })
      const seen = new Set()
      for (const task of tasksFor(spec.id)) {
        for (const rel of task.files) {
          if (seen.has(rel)) continue
          seen.add(rel)
          let text
          try { text = fs.readFileSync(path.join(spec.root, rel), 'utf8') } catch { continue }
          const bytes = Buffer.byteLength(text)
          const readTokens = estTokens(text)
          const sig = pages.get(`file:${rel}`, { budget: SIG_BUDGET })
          const sigTokens = sig.ok ? sig.tokens : null
          rows.push({
            corpus: spec.id, path: rel, bytes, readTokens, sigTokens,
            forcedTokens: sig.ok ? sigTokens + FAULT_EST_TOKENS : null,
          })
        }
      }
    } finally { store.close() }
    for (const suf of ['', '-wal', '-shm']) fs.rmSync(dbPath + suf, { force: true })
  }

  const byThreshold = {}
  const totalRead = rows.reduce((a, r) => a + r.readTokens, 0)
  for (const t of thresholds) {
    const denied = rows.filter((r) => r.bytes >= t && r.sigTokens != null)
    const deniedRead = denied.reduce((a, r) => a + r.readTokens, 0)
    byThreshold[t] = {
      denied: denied.length,
      interceptedShare: totalRead ? deniedRead / totalRead : 0,
      forcedVsRead: denied.reduce((a, r) => a + r.forcedTokens, 0) / (deniedRead || 1),
      overridePenalty: denied.reduce((a, r) => Math.max(a, r.sigTokens + TOOL_OVERHEAD), 0),
      rows: denied.map((r) => ({ ...r, ratio: r.readTokens / r.forcedTokens })),
    }
  }
  return { rows, byThreshold, totalRead }
}

export function renderTable(sim) {
  const lines = ['| threshold | files denied | intercepted share of read tokens | forced/read | max override penalty |', '| --- | --- | --- | --- | --- |']
  for (const [t, v] of Object.entries(sim.byThreshold)) {
    lines.push(`| ${Number(t) / 1024} KB | ${v.denied} | ${(v.interceptedShare * 100).toFixed(0)}% | ${v.forcedVsRead.toFixed(2)} | ${v.overridePenalty} tok |`)
  }
  return lines.join('\n')
}

if (process.argv[1] && url.pathToFileURL(path.resolve(process.argv[1])).href === import.meta.url) {
  const sim = await guardSim()
  process.stdout.write(renderTable(sim) + '\n')
  const t8 = sim.byThreshold[8192]
  const worst = t8.rows.reduce((a, r) => (r.ratio < a.ratio ? r : a), t8.rows[0])
  if (worst) process.stdout.write(`worst 8 KB file: ${worst.path} — read ${worst.readTokens} vs forced ${worst.forcedTokens} (${worst.ratio.toFixed(2)}x)\n`)
}
```
